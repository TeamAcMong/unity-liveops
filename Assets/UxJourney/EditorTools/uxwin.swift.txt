// uxwin — trợ giúp G-UX-JOURNEY: app đứng trước, kích hoạt app, liệt kê cửa sổ của một pid (CGWindowID để screencapture -l).
import Cocoa
let args = CommandLine.arguments
guard args.count >= 2 else { print("usage: uxwin front | activate <pid> | list <pid>"); exit(2) }
switch args[1] {
case "cursor":
    // vị trí con trỏ chuột THẬT (toạ độ màn hình, gốc trên-trái)
    let location = CGEvent(source: nil)?.location ?? .zero
    print("\(Int(location.x)) \(Int(location.y))")
case "warp":
    // Dời con trỏ THẬT ra khỏi cửa sổ hub: con trỏ đậu trong hub làm bật hover và làm bẩn ảnh chụp.
    guard args.count >= 4, let x = Double(args[2]), let y = Double(args[3]) else { exit(2) }
    CGWarpMouseCursorPosition(CGPoint(x: x, y: y))
    CGAssociateMouseAndMouseCursorPosition(1)
    print("ok")
case "front":
    print(NSWorkspace.shared.frontmostApplication?.processIdentifier ?? -1)
case "activate":
    guard args.count >= 3, let pid = Int32(args[2]), let app = NSRunningApplication(processIdentifier: pid) else { print("false"); exit(1) }
    print(app.activate(options: [.activateIgnoringOtherApps]))
case "list":
    guard args.count >= 3, let pid = Int32(args[2]) else { exit(2) }
    let info = CGWindowListCopyWindowInfo([.optionAll], kCGNullWindowID) as? [[String: Any]] ?? []
    for window in info {
        guard let owner = window[kCGWindowOwnerPID as String] as? Int32, owner == pid else { continue }
        let id = window[kCGWindowNumber as String] as? Int ?? 0
        let layer = window[kCGWindowLayer as String] as? Int ?? 0
        let onscreen = window[kCGWindowIsOnscreen as String] as? Bool ?? false
        let alpha = window[kCGWindowAlpha as String] as? Double ?? 0
        let name = window[kCGWindowName as String] as? String ?? ""
        var x = 0.0, y = 0.0, w = 0.0, h = 0.0
        if let bounds = window[kCGWindowBounds as String] as? [String: Any] {
            x = bounds["X"] as? Double ?? 0; y = bounds["Y"] as? Double ?? 0
            w = bounds["Width"] as? Double ?? 0; h = bounds["Height"] as? Double ?? 0
        }
        print("\(id)\t\(layer)\t\(Int(x))\t\(Int(y))\t\(Int(w))\t\(Int(h))\t\(onscreen ? 1 : 0)\t\(alpha)\t\(name)")
    }
case "key":
    // key <pid> <virtualKeyCode> [shift,cmd,alt] — gửi phím THẲNG vào tiến trình (không cần app đứng trước)
    guard args.count >= 4, let pid = Int32(args[2]), let code = UInt16(args[3]) else { exit(2) }
    var flags: CGEventFlags = []
    if args.count >= 5 {
        if args[4].contains("shift") { flags.insert(.maskShift) }
        if args[4].contains("cmd") { flags.insert(.maskCommand) }
        if args[4].contains("alt") { flags.insert(.maskAlternate) }
    }
    let source = CGEventSource(stateID: .combinedSessionState)
    guard let down = CGEvent(keyboardEventSource: source, virtualKey: code, keyDown: true),
          let up = CGEvent(keyboardEventSource: source, virtualKey: code, keyDown: false) else { print("false"); exit(1) }
    down.flags = flags; up.flags = flags
    down.postToPid(pid); usleep(40000); up.postToPid(pid)
    print("true")
case "type":
    guard args.count >= 4, let pid = Int32(args[2]) else { exit(2) }
    let source = CGEventSource(stateID: .combinedSessionState)
    for scalar in args[3].utf16 {
        var character = scalar
        guard let down = CGEvent(keyboardEventSource: source, virtualKey: 0, keyDown: true),
              let up = CGEvent(keyboardEventSource: source, virtualKey: 0, keyDown: false) else { continue }
        down.keyboardSetUnicodeString(stringLength: 1, unicodeString: &character)
        up.keyboardSetUnicodeString(stringLength: 1, unicodeString: &character)
        down.postToPid(pid); usleep(30000); up.postToPid(pid); usleep(30000)
    }
    print("true")
default:
    exit(2)
}
